# Dependencies for Streamlit Community Cloud (dashboard/app.py).
#
# Streamlit Cloud picks requirements.txt over pyproject.toml, so this file is
# what the hosted dashboard installs from. It exists for two reasons:
#
#   1. Pure-Python only. No compiled extensions, so the build can't break on
#      whatever CPython version the platform happens to run. (This bit us
#      once: libsql-experimental ships no cp314 wheel, Streamlit Cloud runs
#      CPython 3.14, pip tried a source build and the Rust step failed. We
#      now reach Turso over its HTTP API instead — see scanner/turso_http.py.)
#   2. Dashboard-only surface. beautifulsoup4 is scanner-only and not needed
#      to render charts.
#
# pyproject.toml stays the source of truth for local dev and GitHub Actions.
# Keep the two in sync when you add something the dashboard imports.

pandas>=3.0,<4.0
plotly>=6.9,<7.0
